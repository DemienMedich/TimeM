cmake_minimum_required(VERSION 3.16)
project(TimeManagementApp VERSION 1.0.0 LANGUAGES CXX)

# Установка стандарта C++
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Включение автоматической обработки MOC, UIC и RCC
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)

# Поиск пакета Qt
find_package(Qt6 COMPONENTS Core Gui Widgets Charts REQUIRED)
if (NOT Qt6_FOUND)
    find_package(Qt5 5.15 COMPONENTS Core Gui Widgets Charts REQUIRED)
endif()

# Директории с заголовочными файлами
include_directories(
    ${CMAKE_CURRENT_SOURCE_DIR}/include
    ${CMAKE_CURRENT_SOURCE_DIR}/include/core
    ${CMAKE_CURRENT_SOURCE_DIR}/include/gui
)

# Исходные файлы основной логики
set(CORE_SOURCES
    src/core/Task.cpp
    src/core/TimeTracker.cpp
    src/core/PomodoroTimer.cpp
    src/core/TaskManager.cpp
    src/core/FileManager.cpp
)

# Исходные файлы GUI
set(GUI_SOURCES
    src/gui/MainWindow.cpp
    src/gui/TaskWidget.cpp
    src/gui/TimeTrackerWidget.cpp
    src/gui/PomodoroWidget.cpp
    src/gui/StatisticsWidget.cpp
    src/gui/SettingsDialog.cpp
)

# Заголовочные файлы основной логики
set(CORE_HEADERS
    include/core/Task.h
    include/core/TimeTracker.h
    include/core/PomodoroTimer.h
    include/core/TaskManager.h
    include/core/FileManager.h
)

# Заголовочные файлы GUI
set(GUI_HEADERS
    include/gui/MainWindow.h
    include/gui/TaskWidget.h
    include/gui/TimeTrackerWidget.h
    include/gui/PomodoroWidget.h
    include/gui/StatisticsWidget.h
    include/gui/SettingsDialog.h
)

# UI файлы
set(UI_FILES
    forms/mainwindow.ui
    forms/taskwidget.ui
    forms/timetrackerwidget.ui
    forms/pomodorowidget.ui
    forms/statisticswidget.ui
    forms/settingsdialog.ui
)

# Файл ресурсов
set(RESOURCE_FILES
    resources/resources.qrc
)

# Файлы переводов
set(TS_FILES
    resources/translations/timemanager_ru.ts
)

# Добавление цели для обновления файлов переводов
if(Qt6_FOUND)
    qt6_add_translation(QM_FILES ${TS_FILES})
else()
    qt5_add_translation(QM_FILES ${TS_FILES})
endif()

# Создание исполняемого файла
add_executable(TimeManagementApp
    src/main.cpp
    ${CORE_SOURCES}
    ${GUI_SOURCES}
    ${CORE_HEADERS}
    ${GUI_HEADERS}
    ${UI_FILES}
    ${RESOURCE_FILES}
    ${QM_FILES}
)

# Линковка с библиотеками Qt
if(Qt6_FOUND)
    target_link_libraries(TimeManagementApp PRIVATE
        Qt6::Core
        Qt6::Gui
        Qt6::Widgets
        Qt6::Charts
    )
else()
    target_link_libraries(TimeManagementApp PRIVATE
        Qt5::Core
        Qt5::Gui
        Qt5::Widgets
        Qt5::Charts
    )
endif()

# Установка
install(TARGETS TimeManagementApp
    RUNTIME DESTINATION bin
)

# Копирование файлов переводов при установке
install(FILES ${QM_FILES} DESTINATION bin/translations)

# Дополнительные флаги компилятора
if(MSVC)
    target_compile_options(TimeManagementApp PRIVATE /W4)
else()
    target_compile_options(TimeManagementApp PRIVATE -Wall -Wextra -pedantic)
endif()
